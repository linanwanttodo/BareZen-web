教程：构建用户管理 API

本教程介绍使用 BareZen 构建完整的用户管理应用 — 从零到带有认证、数据库持久化和测试的运行 API。

我们将构建什么

一个具有以下端点的 REST API：

方法路径描述需要认证
POST/users注册新用户否
POST/login登录并获取会话令牌否
GET/users/{id}按 ID 获取用户是
PUT/users/{id}更新用户是，仅限自己的用户
DELETE/users/{id}删除用户是，仅限自己的用户
GET/health健康检查否

步骤 3：数据库迁移

创建迁移以设置用户表：

package com.example;

import com.barezen.data.Migration;
import java.sql.Connection;
import java.sql.SQLException;

public class CreateUsersTable implements Migration {
    @Override
    public long version() {
        return 1L;
    }

    @Override
    public String description() {
        return "Create users table";
    }

    @Override
    public void apply(Connection conn) throws SQLException {
        try (var stmt = conn.createStatement()) {
            stmt.execute("""
                CREATE TABLE IF NOT EXISTS users (
                    id BIGINT PRIMARY KEY AUTO_INCREMENT,
                    name VARCHAR(255) NOT NULL,
                    email VARCHAR(255) NOT NULL UNIQUE,
                    password_hash VARCHAR(500) NOT NULL,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """);
        }
    }
}

步骤 4：用户仓库

构建用于数据库访问的仓库：

package com.example;

import com.barezen.data.Database;
import java.util.List;
import java.util.Optional;

public class UserRepository {

    private final Database db;

    public UserRepository(Database db) {
        this.db = db;
    }

    public Optional<User> findById(long id) {
        List<User> users = db.query(
            "SELECT id, name, email, password_hash, created_at FROM users WHERE id = ?",
            (rs, rowNum) -> new User(
                rs.getLong("id"),
                rs.getString("name"),
                rs.getString("email"),
                rs.getString("password_hash"),
                rs.getTimestamp("created_at") != null
                    ? rs.getTimestamp("created_at").toLocalDateTime()
                    : null),
            id);
        return users.stream().findFirst();
    }

    public Optional<User> findByEmail(String email) {
        List<User> users = db.query(
            "SELECT id, name, email, password_hash, created_at FROM users WHERE email = ?",
            (rs, rowNum) -> new User(
                rs.getLong("id"),
                rs.getString("name"),
                rs.getString("email"),
                rs.getString("password_hash"),
                rs.getTimestamp("created_at") != null
                    ? rs.getTimestamp("created_at").toLocalDateTime()
                    : null),
            email);
        return users.stream().findFirst();
    }

    public User create(String name, String email, String passwordHash) {
        long id = db.transaction((tx) -> {
            tx.executeUpdate(
                "INSERT INTO users (name, email, password_hash) VALUES (?, ?, ?)",
                name, email, passwordHash);
            return tx.queryForLong("SELECT last_insert_id()");
        });
        return findById(id).orElseThrow();
    }

    public void update(long id, String name, String email) {
        db.executeUpdate(
            "UPDATE users SET name = ?, email = ? WHERE id = ?",
            name, email, id);
    }

    public void delete(long id) {
        db.executeUpdate("DELETE FROM users WHERE id = ?", id);
    }
}

步骤 5：处理程序

为每个端点创建处理程序：

package com.example;

import com.barezen.web.*;
import com.barezen.security.*;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.Map;

public class UserHandlers {

    private final UserRepository repo;
    private final PasswordHasher hasher;
    private final SessionStore sessionStore;

    public UserHandlers(UserRepository repo, PasswordHasher hasher, SessionStore sessionStore) {
        this.repo = repo;
        this.hasher = hasher;
        this.sessionStore = sessionStore;
    }

    // POST /users - 注册
    public Handler register() {
        return (req, params) -> {
            Map<String, String> body = parseJson(req.body());
            String name = body.get("name");
            String email = body.get("email");
            String password = body.get("password");

            if (repo.findByEmail(email).isPresent()) {
                return new Response(409,
                    Headers.of("Content-Type", "application/json"),
                    "{\"error\":\"email already in use\"}".getBytes(StandardCharsets.UTF_8));
            }

            String hash = hasher.hash(password);
            User user = repo.create(name, email, hash);

            return new Response(201,
                Headers.of("Content-Type", "application/json"),
                ("{\"id\":" + user.id() + ",\"name\":\"" + user.name() + "\"}")
                    .getBytes(StandardCharsets.UTF_8));
        };
    }

    // POST /login - 登录
    public Handler login() {
        return (req, params) -> {
            Map<String, String> body = parseJson(req.body());
            String email = body.get("email");
            String password = body.get("password");

            Optional<User> user = repo.findByEmail(email);
            if (user.isEmpty() || !hasher.verify(password, user.get().passwordHash())) {
                return new Response(401,
                    Headers.of("Content-Type", "application/json"),
                    "{\"error\":\"invalid credentials\"}".getBytes(StandardCharsets.UTF_8));
            }

            SessionStore.Session session = sessionStore.create(
                Principal.of(user.get().email()),
                Duration.ofHours(24));

            return new Response(200,
                Headers.of("Content-Type", "application/json")
                    .with("Set-Cookie", "session=" + session.token().value() + "; HttpOnly; Path=/"),
                ("{\"message\":\"logged in\"}").getBytes(StandardCharsets.UTF_8));
        };
    }

    // GET /users/{id} - 获取用户（需要认证）
    public Handler getUser() {
        Handler inner = (req, params) -> {
            long id = Long.parseLong(params.get("id"));
            return repo.findById(id)
                .map(user -> new Response(200,
                    Headers.of("Content-Type", "application/json"),
                    ("{\"id\":" + user.id() + ",\"name\":\"" + user.name()
                     + "\",\"email\":\"" + user.email() + "\"}")
                        .getBytes(StandardCharsets.UTF_8)))
                .orElseGet(() -> new Response(404,
                    Headers.of("Content-Type", "text/plain; charset=utf-8"),
                    "Not Found".getBytes(StandardCharsets.UTF_8)));
        };
        return SecuredHandler.authenticated(inner);
    }

    // PUT /users/{id} - 更新用户（需要认证）
    public Handler updateUser() {
        Handler inner = (req, params) -> {
            long id = Long.parseLong(params.get("id"));
            Map<String, String> body = parseJson(req.body());
            repo.update(id, body.get("name"), body.get("email"));
            return new Response(200,
                Headers.of("Content-Type", "application/json"),
                "{\"message\":\"updated\"}".getBytes(StandardCharsets.UTF_8));
        };
        return SecuredHandler.authenticated(inner);
    }

    // DELETE /users/{id} - 删除用户（需要认证）
    public Handler deleteUser() {
        Handler inner = (req, params) -> {
            long id = Long.parseLong(params.get("id"));
            repo.delete(id);
            return new Response(204, Headers.empty(), null);
        };
        return SecuredHandler.authenticated(inner);
    }

    // 简单 JSON 解析器占位符 - 在生产中使用真实的库
    private Map<String, String> parseJson(byte[] body) {
        // 在生产中，使用 Jackson、Gson 或类似的库
        return Map.of();
    }
}

步骤 6：应用组装

将所有内容连接在一起：

package com.example;

import com.barezen.core.config.Configuration;
import com.barezen.core.di.ServiceAssembly;
import com.barezen.core.kernel.LifecycleOrchestrator;
import com.barezen.data.*;
import com.barezen.http.jdk.JdkHttpServerAdapter;
import com.barezen.security.*;
import com.barezen.web.*;
import org.h2.jdbcx.JdbcDataSource;

import java.net.InetSocketAddress;
import java.nio.file.Path;
import java.time.Duration;
import java.util.Set;

public class Application {

    public static void main(String[] args) throws Exception {
        // 1. 配置
        Configuration config = Configuration.standard(
            Path.of("application.properties"));

        // 2. 数据库
        JdbcDataSource dataSource = new JdbcDataSource();
        dataSource.setURL("jdbc:h2:file:./data/mydb;AUTO_SERVER=TRUE");
        Database db = new Database(dataSource);

        // 3. 运行迁移
        new Migrator(db).migrate(java.util.List.of(new CreateUsersTable()));

        // 4. 安全组件
        PasswordHasher hasher = PasswordHasher.standard();
        SessionStore sessionStore = SessionStore.standard();

        // 5. 服务组装
        ServiceAssembly assembly = ServiceAssembly.builder()
            .instance(Database.class, db)
            .instance(PasswordHasher.class, hasher)
            .instance(SessionStore.class, sessionStore)
            .factory(UserRepository.class, s -> new UserRepository(s.require(Database.class)))
            .factory(UserHandlers.class, s -> new UserHandlers(
                s.require(UserRepository.class),
                s.require(PasswordHasher.class),
                s.require(SessionStore.class)))
            .build();

        // 6. 构建处理程序
        UserHandlers handlers = assembly.require(UserHandlers.class);

        // 7. 构建路由器
        Router router = Router.builder()
            .route(HttpMethod.GET, "/health", (req, params) ->
                new Response(200, Headers.of("Content-Type", "text/plain"), "OK".getBytes()))
            .route(HttpMethod.POST, "/users", handlers.register())
            .route(HttpMethod.POST, "/login", handlers.login())
            .route(HttpMethod.GET, "/users/{id}", handlers.getUser())
            .route(HttpMethod.PUT, "/users/{id}", handlers.updateUser())
            .route(HttpMethod.DELETE, "/users/{id}", handlers.deleteUser())
            .build();

        // 8. HTTP 适配器
        InetSocketAddress address = new InetSocketAddress("0.0.0.0", 8080);
        JdkHttpServerAdapter adapter = new JdkHttpServerAdapter(address, router, config);

        // 9. 生命周期编排器
        LifecycleOrchestrator orchestrator = LifecycleOrchestrator.builder()
            .register("http", adapter)
            .build();

        try {
            orchestrator.start();
            System.out.println("服务器在端口 " + adapter.port().getAsInt() + " 上运行");
            // 阻塞直到关闭
            Thread.currentThread().join();
        } finally {
            orchestrator.stop();
            assembly.close();
        }
    }
}

步骤 7：测试

编写集成测试：

package com.example;

import com.barezen.testkit.*;
import org.junit.jupiter.api.Test;

class UserApiTest {

    @Test
    void registerAndLogin() throws Exception {
        try (TestDatabase testDb = TestDatabase.create()) {
            // 运行迁移
            new com.barezen.data.Migrator(testDb.database())
                .migrate(java.util.List.of(new CreateUsersTable()));

            // 构建组件
            PasswordHasher hasher = PasswordHasher.standard();
            SessionStore sessionStore = SessionStore.standard();
            UserRepository repo = new UserRepository(testDb.database());
            UserHandlers handlers = new UserHandlers(repo, hasher, sessionStore);

            // 构建路由器
            Router router = Router.builder()
                .route(HttpMethod.POST, "/users", handlers.register())
                .route(HttpMethod.POST, "/login", handlers.login())
                .build();

            // 启动服务器
            try (TestServer server = TestServer.start(router)) {
                // 注册
                TestResponse register = server.client()
                    .post("/users", "{\"name\":\"Alice\",\"email\":\"alice@example.com\",\"password\":\"secret\"}");
                Assertions.assertThat(register).statusIs(201);

                // 登录
                TestResponse login = server.client()
                    .post("/login", "{\"email\":\"alice@example.com\",\"password\":\"secret\"}");
                Assertions.assertThat(login).statusIs(200);
            }
        }
    }
}

总结

本教程涵盖了：

  • 领域模型 — 带有 id、name、email、密码哈希的 User 记录
  • 数据库迁移 — CreateUsersTable 迁移
  • 仓库 — 带有 CRUD 操作的 UserRepository
  • 处理程序 — 注册、登录、获取、更新、删除
  • 安全 — 使用 PBKDF2 的密码哈希、会话管理
  • 组装 — ServiceAssembly 连接所有组件
  • 测试 — 使用 TestServer 和 TestDatabase 的集成测试
